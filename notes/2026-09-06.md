def create_session() -> requests.Session:
    session = requests.Session()
    session.headers.update({
        "User-Agent": USER_AGENT,

        if location:
            params["location"] = location
    
        try:
            response = session.get(

                params=params,
                timeout=10

            print(f"[!] Request failed for '{keyword}': {e}")
            continue
    
    return all_results
